def bucket_sort(arr):
    n = len(arr)
    if n <= 1:
        return arr

    # 1. Create n empty buckets
    buckets = [[] for _ in range(n)]

    # 2. Distribute elements into corresponding buckets
    for num in arr:
        bucket_idx = int(n * num)  # Map float [0, 1) to index [0, n-1]
        buckets[bucket_idx].append(num)

    # 3. Sort individual buckets (using insertion/standard sort)
    for bucket in buckets:
        bucket.sort()

    # 4. Concatenate all buckets into the final array
    sorted_arr = []
    for bucket in buckets:
        sorted_arr.extend(bucket)

    return sorted_arr

def quick_sort(arr, start=0, end=None):
    if end is None:
        end = len(arr) - 1
    if start < end:
        pivot = arr[end]
        i = start - 1
        for j in range(start, end):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        arr[i + 1], arr[end] = arr[end], arr[i + 1]
        pivot_idx = i + 1
        quick_sort(arr, start, pivot_idx - 1)
        quick_sort(arr, pivot_idx + 1, end)
    return arr

def merge_sort(arr):
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result

def heap_sort(arr):
    n = len(arr)
    def heapify(heap_size, i):
        largest = i
        l = 2 * i + 1
        r = 2 * i + 2
        if l < heap_size and arr[l] > arr[largest]:
            largest = l
        if r < heap_size and arr[r] > arr[largest]:
            largest = r
        if largest != i:
            arr[i], arr[largest] = arr[largest], arr[i]
            heapify(heap_size, largest)

    for i in range(n // 2 - 1, -1, -1):
        heapify(n, i)
    for i in range(n - 1, 0, -1):
        arr[0], arr[i] = arr[i], arr[0]
        heapify(i, 0)
    return arr

from collections import deque

def bfs_graph(graph, start_node):
    visited = set([start_node])
    queue = deque([start_node])
    traversal_order = []

    while queue:
        node = queue.popleft()
        traversal_order.append(node)

        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

    return traversal_order

def solve_n_queens(n):
    # N-Queens via backtracking, undoing placement on dead ends
    result = []
    cols = set()
    diag1 = set()
    diag2 = set()
    placement = []

    def backtrack(row):
        if row == n:
            result.append(placement[:])
            return
        for col in range(n):
            if col in cols or (row - col) in diag1 or (row + col) in diag2:
                continue
            cols.add(col)
            diag1.add(row - col)
            diag2.add(row + col)
            placement.append(col)
            backtrack(row + 1)
            placement.pop()
            cols.remove(col)
            diag1.remove(row - col)
            diag2.remove(row + col)

    backtrack(0)
    return result

function permute(arr) {
  // Generate all permutations via backtracking with explicit undo
  const result = [];
  const path = [];
  const used = new Array(arr.length).fill(false);

  function backtrack() {
    if (path.length === arr.length) {
      result.push([...path]);
      return;
    }
    for (let i = 0; i < arr.length; i++) {
      if (used[i]) continue;
      used[i] = true;
      path.push(arr[i]);
      backtrack();
      path.pop();
      used[i] = false;
    }
  }

  backtrack();
  return result;
}

def combination_sum(candidates, target):
    result = []
    path = []

    def backtrack(start, remaining):
        if remaining == 0:
            result.append(path[:])
            return
        for i in range(start, len(candidates)):
            if candidates[i] <= remaining:
                path.append(candidates[i])
                backtrack(i, remaining - candidates[i])
                path.pop()

    backtrack(0, target)
    return result

function exist(board, word) {
  const R = board.length;
  const C = board[0].length;

  function searchWord(r, c, k) {
    if (k === word.length) return true;
    if (r < 0 || r >= R || c < 0 || c >= C || board[r][c] !== word[k]) {
      return false;
    }

    const temp = board[r][c];
    board[r][c] = "#";

    const found =
      searchWord(r + 1, c, k + 1) ||
      searchWord(r - 1, c, k + 1) ||
      searchWord(r, c + 1, k + 1) ||
      searchWord(r, c - 1, k + 1);

    board[r][c] = temp;
    return found;
  }

  for (let r = 0; r < R; r++) {
    for (let c = 0; c < C; c++) {
      if (searchWord(r, c, 0)) return true;
    }
  }
  return false;
}

def exist(board, word):
    R, C = len(board), len(board[0])

    def search_word(r, c, k):
        if k == len(word):

        if r < 0 or r >= R or c < 0 or c >= C or board[r][c] != word[k]:
            return False
        
        temp = board[r][c]
        board[r][c] = "#"  # mark visited
        
        found = (
            search_word(r + 1, c, k + 1) or
            search_word(r - 1, c, k + 1) or
            search_word(r, c + 1, k + 1) or
            search_word(r, c - 1, k + 1)
        )
        
        board[r][c] = temp  # backtrack unmark
        return found

    return any(search_word(r, c, 0) for r in range(R) for c in range(C))

def hamiltonian_cycle(graph, n):
    path = [-1] * n
    path[0] = 0  # Start at node 0
    visited = [False] * n
    visited[0] = True

    def backtrack(pos):
        # Base case: All vertices are included in path
        if pos == n:
            # Check if there is an edge from the last vertex to the first vertex
            return graph[path[pos - 1]][path[0]] == 1

        for v in range(1, n):
            # Check if vertex v can be added to Hamiltonian Cycle
            if graph[path[pos - 1]][v] == 1 and not visited[v]:
                path[pos] = v
                visited[v] = True

                if backtrack(pos + 1):
                    return True

                # Backtrack / Undo state
                path[pos] = -1
                visited[v] = False

        return False

    if backtrack(1):
        return path + [path[0]]
    return []

    [0,  10, 15, 20, 10],  # Edges from Node 0
    [10,  0, 35, 25, 12],  # Edges from Node 1
    [15, 35,  0, 30,  8],  # Edges from Node 2
    [20, 25, 30,  0, 14],  # Edges from Node 3
    [10, 12,  8, 14,  0]   # Edges from Node 4

def tsp(graph, n):
    visited = [False] * n
    visited[0] = True
    min_cost = [float('inf')]

    def backtrack(curr, count, cost):
        if count == n and graph[curr][0] > 0:
            min_cost[0] = min(min_cost[0], cost + graph[curr][0])
            return

        for nxt in range(n):
            if not visited[nxt] and graph[curr][nxt] > 0:
                if cost + graph[curr][nxt] < min_cost[0]:  # Pruning branch
                    visited[nxt] = True
                    backtrack(nxt, count + 1, cost + graph[curr][nxt])
                    visited[nxt] = False

    backtrack(0, 1, 0)
    return min_cost[0]

print("Minimum TSP Tour Cost:", tsp(graph, n))

def max_sum_subarray(arr, k):
    # Fixed-size sliding window of size k
    window_sum = sum(arr[:k])
    max_sum = window_sum
    
    for i in range(k, len(arr)):
        window_sum += arr[i] - arr[i - k]
        max_sum = max(max_sum, window_sum)
        
    return max_sum

def length_of_longest_substring(s):
    # Variable-size sliding window with hash set
    seen = set()
    left = 0
    max_len = 0
    
    for right in range(len(s)):
        while s[right] in seen:
            seen.remove(s[left])
            left += 1
        seen.add(s[right])
        max_len = max(max_len, right - left + 1)
        
    return max_len

    left = 0
    right = len(numbers) - 1

def climb_stairs(n, memo=None):
    if memo is None:
        memo = {}
    if n <= 1:
        return 1
    if n in memo:
        return memo[n]

def coin_change(target, arr):
    dp = [float('inf')] * (target + 1)
    dp[0] = 0
    for i in range(1, target + 1):
        for coin in arr:
            if coin <= i:

def lcs(arr, arr2):
    n = len(arr)
    m = len(arr2)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if arr[i - 1] == arr2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    return dp[n][m]

def knapsack(arr, arr2, target):
    n = len(arr)
    dp = [[0] * (target + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(target + 1):
            dp[i][j] = dp[i - 1][j]
            if arr[i - 1] <= j:
                candidate = dp[i - 1][j - arr[i - 1]] + arr2[i - 1]
                dp[i][j] = max(dp[i][j], candidate)
    return dp[n][target]

function sortList(head) {
    if (!head || !head.next) return head;

    // 1. Find middle with Fast & Slow pointers
    let slow = head;
    let fast = head.next;
    while (fast && fast.next) {
        slow = slow.next;
        fast = fast.next.next;
    }

    // 2. Split list into two halves
    let mid = slow.next;
    slow.next = null;

    // 3. Recursive sort & merge
    let left = sortList(head);
    let right = sortList(mid);
    return merge(left, right);
}

function merge(l1, l2) {
    let dummy = { val: 0, next: null };
    let curr = dummy;
    while (l1 && l2) {
        if (l1.val <= l2.val) {
            curr.next = l1;
            l1 = l1.next;
        } else {
            curr.next = l2;
            l2 = l2.next;
        }
        curr = curr.next;
    }
    curr.next = l1 || l2;
    return dummy.next;
}

function horspool(text, pattern) {
    const m = pattern.length;
    const n = text.length;
    const shiftTable = {};

    // 1. Build Bad-Symbol Shift Table
    for (let i = 0; i < m - 1; i++) {
        shiftTable[pattern[i]] = m - 1 - i;
    }

    // 2. Search text
    let i = 0;
    while (i <= n - m) {
        let k = m - 1;
        while (k >= 0 && pattern[k] === text[i + k]) {
            k--;
        }
        if (k < 0) return i; // Found occurrence
        const shift = shiftTable[text[i + m - 1]] || m;
        i += shift;
    }
    return -1;
}

# Horspool's String Matching Algorithm
def horspool(text: str, pattern: str) -> int:
    m = len(pattern)
    n = len(text)
    if m > n or m == 0:
        return -1

    shift_table = {}
    for i in range(m - 1):
        shift_table[pattern[i]] = m - 1 - i

    # 2. Search Text
    i = 0
    while i <= n - m:
        k = m - 1
        # Compare right-to-left
        while k >= 0 and pattern[k] == text[i + k]:
            k -= 1

        if k < 0:
            return i  # Match found at index i

        # Shift based on text character aligned with pattern's last character

def three_sum(nums: list[int]) -> list[list[int]]:
    nums.sort()
    res = []

    for i in range(len(nums) - 2):
        if i > 0 and nums[i] == nums[i - 1]:
            continue

        left = i + 1
        right = len(nums) - 1

        while left < right:
            total = nums[i] + nums[left] + nums[right]

            if total == 0:
                res.append([nums[i], nums[left], nums[right]])

                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1

                left += 1
                right -= 1
            elif total < 0:
                left += 1
            else:
                right -= 1

    return res

def binary_search(arr, target):
    # Standard Binary Search on a sorted array
    left, right = 0, len(arr) - 1

    while left <= right:
        mid = left + (right - left) // 2  # Prevents potential integer overflow

        if arr[mid] == target:
            return mid  # Target found, return index
        elif arr[mid] < target:
            left = mid + 1  # Search right half
        else:
            right = mid - 1  # Search left half

    return -1  # Target not found

# 1. Quick Sort
def quick_sort(arr, start=0, end=None):
    if end is None:
        end = len(arr) - 1
    if start < end:
        pivot = arr[end]
        i = start - 1
        for j in range(start, end):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        arr[i + 1], arr[end] = arr[end], arr[i + 1]
        pivot_idx = i + 1
        quick_sort(arr, start, pivot_idx - 1)
        quick_sort(arr, pivot_idx + 1, end)
    return arr

# 2. Merge Sort
def merge_sort(arr):
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result

def insertion_sort(arr):

        j = i - 1

            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    return arr

# 4. Shell Sort
def shell_sort(arr):
    n = len(arr)
    gap = n // 2
    while gap > 0:
        for i in range(gap, n):
            tmp = arr[i]
            j = i
            while j >= gap and arr[j - gap] > tmp:
                arr[j] = arr[j - gap]
                j -= gap
            arr[j] = tmp
        gap //= 2
    return arr

# 5. Heap Sort
def heap_sort(arr):
    n = len(arr)
    def heapify(heap_size, i):
        largest = i
        l = 2 * i + 1
        r = 2 * i + 2
        if l < heap_size and arr[l] > arr[largest]:
            largest = l
        if r < heap_size and arr[r] > arr[largest]:
            largest = r
        if largest != i:
            arr[i], arr[largest] = arr[largest], arr[i]
            heapify(heap_size, largest)

    for i in range(n // 2 - 1, -1, -1):
        heapify(n, i)
    for i in range(n - 1, 0, -1):
        arr[0], arr[i] = arr[i], arr[0]
        heapify(i, 0)
    return arr

# 6. Counting Sort
def counting_sort(arr):
    if not arr:
        return arr
    maxv, minv = max(arr), min(arr)
    span = maxv - minv + 1
    count = [0] * span
    for num in arr:
        count[num - minv] += 1
    for i in range(1, span):
        count[i] += count[i - 1]
    result = [0] * len(arr)
    for num in reversed(arr):
        count[num - minv] -= 1
        result[count[num - minv]] = num
    return result

# 7. Radix Sort
def _radix_counting_pass(arr, exp):
    n = len(arr)
    output = [0] * n
    count = [0] * 10
    for i in range(n):
        digit = (arr[i] // exp) % 10
        count[digit] += 1
    for i in range(1, 10):
        count[i] += count[i - 1]
    for i in range(n - 1, -1, -1):
        digit = (arr[i] // exp) % 10
        count[digit] -= 1
        output[count[digit]] = arr[i]
    return output

def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        swapped = False
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swapped = True
        if not swapped:
            break

def eval_rpn(tokens: list[str]) -> int:
    stack = []

    for token in tokens:
        if token == "+":
            stack.append(stack.pop() + stack.pop())
        elif token == "-":
            b, a = stack.pop(), stack.pop()
            stack.append(a - b)
        elif token == "*":
            stack.append(stack.pop() * stack.pop())
        elif token == "/":
            b, a = stack.pop(), stack.pop()
            stack.append(int(a / b))
        else:
            stack.append(int(token))

    return stack[0]

import heapq


def prim_mst(n: int, edges: list[list[int]]) -> tuple[int, list[tuple[int, int, int]]]:
    adj = {i: [] for i in range(n)}
    for u, v, w in edges:
        adj[u].append((w, v))
        adj[v].append((w, u))

    min_heap = [(0, 0, -1)]
    visited = [False] * n
    total_weight = 0
    mst_edges = []

    while min_heap:
        weight, u, parent = heapq.heappop(min_heap)

        if visited[u]:
            continue

        visited[u] = True
        total_weight += weight

        if parent != -1:
            mst_edges.append((parent, u, weight))

        for edge_weight, v in adj[u]:
            if not visited[v]:
                heapq.heappush(min_heap, (edge_weight, v, u))

    return total_weight, mst_edges

        for i in range(index, len(nums)):
            path.append(nums[i])
            backtrack(i + 1, path)

def topological_sort_dfs(v: int, adj: list[list[int]]) -> list[int]:
    visited = [0] * v  # 0: Unvisited, 1: Visiting (in stack), 2: Visited
    stack = []

    def dfs(node: int) -> bool:
        visited[node] = 1  # Mark current node as visiting

        for neighbor in adj[node]:
            if visited[neighbor] == 1:
                return True  # Cycle detected
            if visited[neighbor] == 0:
                if dfs(neighbor):
                    return True

        visited[node] = 2  # Mark current node as fully visited
        stack.append(node)  # Post-order placement
        return False

    for i in range(v):
        if visited[i] == 0:
            if dfs(i):
                return []  # Return empty list if graph has a cycle (not a DAG)

    return stack[::-1]

def lowest_common_ancestor(root, p, q):
    if root is None:
        return None

    if root == p or root == q:
        return root

    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)

    if left and right:

    return left if left else right

def longest_increasing_subsequence(nums):
    if not nums:
        return 0

    dp = [1] * len(nums)

    for i in range(len(nums)):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)

def stock_span(prices):
    stack = []
    spans = []

    for i, price in enumerate(prices):
        while stack and prices[stack[-1]] <= price:
            stack.pop()

        if not stack:
            span = i + 1
        else:
            span = i - stack[-1]

        spans.append(span)
        stack.append(i)

def word_break(s, word_dict):
    dp = [False] * (len(s) + 1)
    dp[0] = True

    for i in range(1, len(s) + 1):
        for j in range(i):
            if dp[j] and s[j:i] in word_dict:
                dp[i] = True
                break

    return dp[len(s)]

House Robber II
def rob_linear(nums):
    prev2 = 0
    prev1 = 0

    for money in nums:
        current = max(prev1, prev2 + money)
        prev2 = prev1
        prev1 = current

    return prev1


def house_robber_ii(nums):
    if not nums:
        return 0

    if len(nums) == 1:
        return nums[0]

    return max(
        rob_linear(nums[:-1]),
        rob_linear(nums[1:])
    )


nums = [2, 3, 2]

print(house_robber_ii(nums))

def interval_intersection(first_list, second_list):
    i = 0
    j = 0
    result = []

    while i < len(first_list) and j < len(second_list):
        start = max(first_list[i][0], second_list[j][0])
        end = min(first_list[i][1], second_list[j][1])

        if start <= end:
            result.append([start, end])

        if first_list[i][1] < second_list[j][1]:
            i += 1
        else:
            j += 1

    return result

class Node:

        self.next = None


def insert_at_position(head, value, position):
    new_node = Node(value)

    if position == 0:
        new_node.next = head
        return new_node

    current = head

    for _ in range(position - 1):
        if current is None:
            return head
        current = current.next

    if current is None:
        return head

    new_node.next = current.next
    current.next = new_node

    return head

def delete_value(head, target):
    if head is None:
        return None

    if head.data == target:
        return head.next

    prev = head
    current = head.next

    while current is not None:
        if current.data == target:
            prev.next = current.next
            return head

        prev = current
        current = current.next

    return head

    if root is None:
        return []

    stack1 = [root]
    stack2 = []
    result = []

    while stack1:
        node = stack1.pop()
        stack2.append(node)

        if node.left:
            stack1.append(node.left)

        if node.right:
            stack1.append(node.right)

    while stack2:
        result.append(stack2.pop().val)

    return result

### Level Order - BFS
from collections import deque


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


def level_order(root):
    if root is None:
        return []

    result = []
    queue = deque([root])

    while queue:
        level = []

        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.val)

            if node.left:
                queue.append(node.left)

            if node.right:
                queue.append(node.right)

        result.append(level)

    return result


root = TreeNode(
    1,
    TreeNode(2, TreeNode(4), TreeNode(5)),
    TreeNode(3, TreeNode(6), TreeNode(7))
)

print(level_order(root))

import heapq
from collections import Counter


def top_k_frequent(nums, k):
    frequency = Counter(nums)

    heap = []

    for num, count in frequency.items():
        heapq.heappush(heap, (count, num))

        if len(heap) > k:
            heapq.heappop(heap)

    return [num for count, num in heap]


nums = [1, 1, 1, 2, 2, 3]
k = 2

print("Top K Frequent:", top_k_frequent(nums, k))
# Output: [2, 1]

import heapq


def kth_largest(nums, k):
    heap = []

    for num in nums:
        heapq.heappush(heap, num)

        if len(heap) > k:

def two_sum(nums, target):
    seen = {}

    for i, num in enumerate(nums):
        complement = target - num

        if complement in seen:
            return [seen[complement], i]

        seen[num] = i

    return []


nums = [2, 7, 11, 15]
target = 9

print("Two Sum:", two_sum(nums, target))
